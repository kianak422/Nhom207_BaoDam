// src/pages/CreatePost.jsx

    import { useState, useEffect, useRef } from 'react';
    import { useNavigate } from 'react-router-dom';
    import { useAuth } from '../contexts/AuthContext';
   import { useCategories } from '../contexts/CategoryContext';
    import styles from '../components/CreatePost.module.css';

  import { createPost, saveDraft } from '../services/postService'; 
    import { fetchProvinces, fetchDistricts, fetchWards } from '../services/locationService';
    import { numberToVietnameseWords, formatNumber } from '../utils/formatters';
    import { MdKeyboardArrowDown } from 'react-icons/md';

    const SpecificFields = ({ fields, formData, handleChange, errors, isDescriptionGuideVisible, onFocus, onBlur }) => {
      return fields.map((field) => {
          const commonProps = {
              name: field.name,
              id: field.name,
              onChange: handleChange,
              value: formData[field.name] || '',
              className: errors[field.name] ? `${styles.input} ${styles.inputError}` : styles.input,
              required: field.validation?.required,
              "data-testid": `post-${field.name}-input` // TỰ ĐỘNG SINH ID: post-price-input, post-title-input...
          };

          const renderField = () => {
              switch (field.type) {
                  case "price":
                      return <input type="text" placeholder={field.placeholder} {...commonProps} value={formData.displayPrice || ''} name="price" />;
                  case "number":
                      return <input type="number" placeholder={field.placeholder} {...commonProps} />;
                  case "select":
                      return <select {...commonProps} className={`${styles.input} ${styles.select}`}><option value="">-- Chọn --</option>{field.options.map(opt => <option key={opt} value={opt}>{opt}</option>)}</select>;
                  case "textarea":
                      return <textarea rows="5" placeholder={field.placeholder} {...commonProps} onFocus={() => onFocus(field.name)} onBlur={() => onBlur(field.name)} />;
                  default:
                      return <input type="text" placeholder={field.placeholder} {...commonProps} />;
              }
          };

          return (
              <div key={field.name} className={styles.inputGroup}>
                  <label htmlFor={field.name} className={styles.label}>{field.label}{field.validation?.required && ' *'}</label>
                  {renderField()}
                  {field.type === 'price' && formData.rawPrice && formData.rawPrice > 0 && (
                      <p className={styles.priceInWords}>{numberToVietnameseWords(Number(formData.rawPrice))}</p>
                  )}
                  {errors[field.name] && <p className={styles.errorText}>{errors[field.name]}</p>}

                  {field.name === 'description' && (
                      <>
                          <p className={styles.charCounter}>{ (formData.description || '').length }/1500 kí tự</p>
                          {isDescriptionGuideVisible && (
                              <div className={styles.guideBox}>
                                  <h4 style={{ color: "white", backgroundColor: "#007bff", padding: "5px 10px", borderRadius: "4px", margin: "0 0 10px 0", fontSize: "1rem" }}>Không cho phép</h4>
                                  <ul style={{ margin: 0, paddingLeft: "20px" }}>
                                      <li>Sản phẩm cấm, hạn chế hoặc giả/nhái.</li>
                                      <li>Thông tin trùng lặp với tin cũ.</li>
                                      <li>Chứa số điện thoại, website, hoặc các từ ngữ không phù hợp.</li>
                                  </ul>
                              </div>
                          )}
                      </>
                  )}
              </div>
          );
      });
  };

const CategorySelector = ({ onSelect }) => {
    const { categories, loading } = useCategories(); 

    if (loading) {
        return <div>Đang tải danh mục...</div>;
    }

    return (
        <div className={styles.categorySelectorContainer}>
            <h1 style={{ textAlign: "center", marginBottom: "2rem" }}>Bạn muốn đăng tin về gì?</h1>
            {categories.map((cat) => (
                <div key={cat.value} className={styles.categoryGroup}>
                    <h2 className={styles.categoryTitle}>{cat.label}</h2>
                    <ul className={styles.subcategoryList}>
                        {cat.subcategories.map((sub) => (
                            <li 
                                key={sub.value} 
                                className={styles.subcategoryItem} 
                                onClick={() => onSelect(cat, sub)}
                                data-testid={`category-${cat.value}-${sub.value}`} // Gắn ID cho danh mục để dễ test auto click
                            >
                                {sub.label}
                            </li>
                        ))}
                    </ul>
                </div>
            ))}
        </div>
    );
};


    function CreatePost() {
        const navigate = useNavigate();
        const { userProfile } = useAuth();
        const fileInputRef = useRef(null);
        const formRef = useRef(null); 
        const [draftLoading, setDraftLoading] = useState(false); 

        const [step, setStep] = useState(1);
        const [selectedCategory, setSelectedCategory] = useState(null);
        const [selectedSubcategory, setSelectedSubcategory] = useState(null);

        const [formData, setFormData] = useState({});
      const [errors, setErrors] = useState({});
        const [imageFiles, setImageFiles] = useState([]);
        const [previewUrls, setPreviewUrls] = useState([]);
        const [loading, setLoading] = useState(false);
        const [isDescriptionGuideVisible, setIsDescriptionGuideVisible] = useState(false);
        const [provinces, setProvinces] = useState([]);
        const [districts, setDistricts] = useState([]);
        const [wards, setWards] = useState([]);
        const [selectedProvinceCode, setSelectedProvinceCode] = useState(null);
        const [selectedDistrictCode, setSelectedDistrictCode] = useState(null);
        const [isPostTypeModalOpen, setIsPostTypeModalOpen] = useState(false);
        const [selectedPostType, setSelectedPostType] = useState({ value: 'free', label: 'Đăng tin thường (Miễn phí)' });
        const MAX_IMAGES = 12;

        useEffect(() => {
            fetchProvinces().then(setProvinces);
        }, []);

        useEffect(() => {
            if (!selectedProvinceCode) return;
            setDistricts([]);
            fetchDistricts(selectedProvinceCode).then(setDistricts);
        }, [selectedProvinceCode]);

        useEffect(() => {
            if (!selectedDistrictCode) return;
            setWards([]);
            fetchWards(selectedDistrictCode).then(setWards);
        }, [selectedDistrictCode]);

        useEffect(() => {
            return () => previewUrls.forEach((url) => URL.revokeObjectURL(url));
        }, [previewUrls]);

        
        const handleChange = (e) => {
            const { name, value } = e.target;
            if (name === "price") {
                const rawValue = value.replace(/\D/g, "");
                setFormData((prev) => ({ ...prev, rawPrice: rawValue, displayPrice: formatNumber(rawValue) }));
            } else {
                setFormData({ ...formData, [name]: value });
            }
        };
        
        const handleProvinceChange = (e) => {
          const provinceCode = e.target.value;
          const provinceName = provinceCode ? e.target.options[e.target.selectedIndex].text : "";
          setSelectedProvinceCode(provinceCode);
          setFormData(prev => ({ 
              ...prev, 
              province: provinceName, 
              provinceCode: provinceCode, 
              district: "", 
              districtCode: "", 
              ward: "" 
          }));
          setSelectedDistrictCode(null);
      };

  const handleDistrictChange = (e) => {
      const districtCode = e.target.value;
      const districtName = districtCode ? e.target.options[e.target.selectedIndex].text : "";
      setSelectedDistrictCode(districtCode);

      setFormData(prev => ({ 
          ...prev, 
          district: districtName, 
          districtCode: districtCode,
          ward: "" 
      }));
  };

        const handleWardChange = (e) => {
            const wardName = e.target.value ? e.target.options[e.target.selectedIndex].text : "";
            setFormData({ ...formData, ward: wardName });
        };

        const addFilesToState = (files) => {
            const validFiles = Array.from(files).filter((file) => file.type.startsWith("image/"));
            if (imageFiles.length + validFiles.length > MAX_IMAGES) {
                alert(`Bạn chỉ được đăng tối đa ${MAX_IMAGES} ảnh.`);
                const remainingSlots = MAX_IMAGES - imageFiles.length;
                const filesToAdd = validFiles.slice(0, remainingSlots);
                setImageFiles((prev) => [...prev, ...filesToAdd]);
                setPreviewUrls((prev) => [...prev, ...filesToAdd.map((file) => URL.createObjectURL(file))]);
            } else {
                setImageFiles((prev) => [...prev, ...validFiles]);
                setPreviewUrls((prev) => [...prev, ...validFiles.map((file) => URL.createObjectURL(file))]);
            }
        };
        
        const handleImageChange = (e) => { addFilesToState(e.target.files); e.target.value = null; };
        const handleRemoveImage = (indexToRemove) => {
            URL.revokeObjectURL(previewUrls[indexToRemove]);
            setImageFiles((prev) => prev.filter((_, i) => i !== indexToRemove));
            setPreviewUrls((prev) => prev.filter((_, i) => i !== indexToRemove));
        };

        const handleDragOver = (e) => { e.preventDefault(); e.stopPropagation(); e.currentTarget.style.borderColor = "#ffba00"; };
        const handleDragLeave = (e) => { e.preventDefault(); e.stopPropagation(); e.currentTarget.style.borderColor = "#e0e0e0"; };
        const handleDrop = (e) => { e.preventDefault(); e.stopPropagation(); e.currentTarget.style.borderColor = "#e0e0e0"; addFilesToState(e.dataTransfer.files); };

     const handleCategorySelect = (category, subcategory) => {
        setSelectedCategory(category);
        setSelectedSubcategory(subcategory);

        let initialFormData = {
            category: category.value,
            subcategory: subcategory.value,
            ...category.specificFields.reduce((acc, field) => ({ ...acc, [field.name]: "" }), {}),
        };

        if (userProfile && userProfile.lastUsedAddress) {
            console.log("Phát hiện địa chỉ đã lưu, đang tự động điền...");
            const lastAddress = userProfile.lastUsedAddress;
            
            initialFormData = {
                ...initialFormData,
                province: lastAddress.province,
                district: lastAddress.district,
                ward: lastAddress.ward,
                streetAddress: lastAddress.streetAddress || '',
                provinceCode: lastAddress.provinceCode, 
                districtCode: lastAddress.districtCode,
            };

            setSelectedProvinceCode(lastAddress.provinceCode);
            setSelectedDistrictCode(lastAddress.districtCode);
        }

        setFormData(prev => ({ 
            ...prev, 
            ...initialFormData 
        }));
        
        setErrors({}); 
        setStep(2); 
    };

        const handleSelectPostType = (type) => { setSelectedPostType(type); setIsPostTypeModalOpen(false); };

       const validateForm = () => {
        const newErrors = {};
        if (!selectedCategory) return false;

        selectedCategory.specificFields.forEach(field => {
            if (field.name === 'price') return; 
            const rules = field.validation;
            const value = formData[field.name] || '';

            if (rules?.required && value.trim() === '') {
                newErrors[field.name] = `${field.label} là bắt buộc.`;
            } else if (rules?.minLength && value.length < rules.minLength) {
                newErrors[field.name] = `${field.label} phải có ít nhất ${rules.minLength} ký tự.`;
            }
        });

        const priceField = selectedCategory.specificFields.find(f => f.name === 'price');
        if (priceField?.validation) {
            const rules = priceField.validation;
            const priceValue = Number(formData.rawPrice);
            if (rules.required && (!formData.rawPrice || priceValue === 0)) {
                newErrors.price = rules.errorMessage || 'Giá là bắt buộc.';
            } else if (rules.min && priceValue < rules.min) {
                newErrors.price = rules.errorMessage || `Giá phải lớn hơn ${formatNumber(rules.min)}đ.`;
            } else if (rules.max && priceValue > rules.max) {
                newErrors.price = rules.errorMessage || `Giá phải nhỏ hơn ${formatNumber(rules.max)}đ.`;
            }
        }
        
        if (!formData.province) newErrors.province = 'Vui lòng chọn Tỉnh/Thành phố.';
        if (!formData.district) newErrors.district = 'Vui lòng chọn Quận/Huyện.';
        if (!formData.ward) newErrors.ward = 'Vui lòng chọn Phường/Xã.';
        
        if (imageFiles.length === 0) {
            newErrors.images = "Vui lòng đăng ít nhất 1 ảnh.";
        }

        setErrors(newErrors);
        return Object.keys(newErrors).length === 0;
    };

    const handleSaveDraft = async () => {
          const isFormValid = validateForm();
          if (!isFormValid) {
              return; 
          }

          setDraftLoading(true);
          try {
              await saveDraft(formData, imageFiles, userProfile);
              alert("Đã lưu tin vào bản nháp thành công!");
              navigate('/quan-ly-tin');
          } catch (error) {
              console.error("Lỗi khi lưu nháp:", error);
              alert(`Lưu nháp thất bại: ${error.message}`);
          } finally {
              setDraftLoading(false);
          }
      };

  const handlePreview = () => {
          const isFormValid = validateForm();
          if (!isFormValid) {
              return; 
          }
          
          const previewData = {
              ...formData,
              price: Number(formData.rawPrice) || 0,
              imageUrls: previewUrls,
              authorDisplayName: userProfile.displayName,
              authorAvatarUrl: userProfile.avatarUrl,
              authorPhoneNumber: userProfile.phoneNumber,
              authorId: userProfile.uid,
              createdAt: new Date(),
              id: 'preview-id',
          };

          sessionStorage.setItem('postPreviewData', JSON.stringify(previewData));
          window.open('/xem-truoc', '_blank');
      };


      const handleSubmit = async (e) => {
    e.preventDefault();
    if (!validateForm()) {
        const firstErrorField = document.querySelector(`.${styles.inputError}`);
        if (firstErrorField) {
            firstErrorField.scrollIntoView({ behavior: 'smooth', block: 'center' });
        }
        return;
    }

     setLoading(true);
    try {
        const wantsToPay = selectedPostType.value === 'paid';
        const docRef = await createPost(formData, imageFiles, userProfile, wantsToPay);
        
        if (!userProfile.isVip && wantsToPay) {
            alert("Đã tạo tin chờ thanh toán! Giờ hãy hoàn tất để tin được đẩy top.");
            navigate('/quan-ly-tin'); 
        } else {
            alert("Đăng tin thành công!");
            navigate("/");
        }

    } catch (error) {
        console.error("Lỗi toàn cục khi đăng tin:", error);
        alert(`Đăng tin thất bại: ${error.message}`);
    } finally {
        setLoading(false);
    }
};

      
      
      if (step === 1) {
          return <CategorySelector onSelect={handleCategorySelect} />;
      }

    return (
      <>
          {isPostTypeModalOpen && (
              <div className={styles.modalOverlay} onClick={() => setIsPostTypeModalOpen(false)}>
                  <div className={styles.modalContent} onClick={(e) => e.stopPropagation()}>
                      <button className={styles.closeModalButton} onClick={() => setIsPostTypeModalOpen(false)}>&times;</button>
                      <h3>Chọn lượt đăng tin từ</h3>
                      <div 
                        className={styles.postTypeOption} 
                        onClick={() => handleSelectPostType({ value: 'free', label: 'Đăng tin thường (Miễn phí)' })}
                        data-testid="post-type-free-option"
                      >
                          <h4>Đăng tin thường (Miễn phí)</h4>
                          <p>Loại tin: Tin thường - 60 ngày</p>
                      </div>
                      <div 
                        className={styles.postTypeOption} 
                        onClick={() => handleSelectPostType({ value: 'paid', label: 'Đăng tin trả phí' })}
                        data-testid="post-type-paid-option"
                      >
                          <h4>Đăng tin trả phí</h4>
                          <p>Trả phí đăng tin cho mỗi lượt đăng tin</p>
                      </div>
                      <button 
                        className={styles.applyButton} 
                        onClick={() => setIsPostTypeModalOpen(false)}
                        data-testid="post-type-apply-button"
                      >Áp dụng</button>
                  </div>
              </div>
          )}

          {/* --- FORM CHÍNH --- */}
          <div className={styles.formContainer}>
              <button onClick={() => setStep(1)} className={styles.backButton}>
                  &larr; Chọn lại danh mục
              </button>
              <h1 style={{ textAlign: "center" }}>
                  {`Đăng tin mục "${selectedSubcategory?.label}"`}
              </h1>
              <form ref={formRef} onSubmit={handleSubmit} noValidate>
                  <div className={styles.formGrid}>
                      {/* CỘT TRÁI ẢNH */}
                      <div className={styles.imageColumn}>
                          <h3 className={styles.sectionTitle}>Hình ảnh ({imageFiles.length}/{MAX_IMAGES})</h3>
                          <div 
                            className={styles.imageUploadSection} 
                            onDragOver={handleDragOver} 
                            onDragLeave={handleDragLeave} 
                            onDrop={handleDrop} 
                            onClick={() => fileInputRef.current.click()}
                            data-testid="post-image-upload-section"
                          >
                              <span className={styles.imageUploadIcon}>📸</span>
                              <p className={styles.imageUploadText}>Kéo & Thả ảnh vào đây hoặc click để chọn ảnh</p>
                              <input 
                                type="file" 
                                className={styles.imageInput} 
                                onChange={handleImageChange} 
                                multiple 
                                accept="image/*" 
                                ref={fileInputRef} 
                                data-testid="post-image-input" 
                              />
                          </div>
                          {previewUrls.length > 0 && (
                              <div className={styles.previewImagesContainer}>
                                  {previewUrls.map((url, index) => (
                                      <div key={index} className={styles.previewImageWrapper}>
                                          <img src={url} alt={`Xem trước ${index + 1}`} className={styles.previewImage} />
                                          <button type="button" className={styles.removeImageButton} onClick={(e) => { e.stopPropagation(); handleRemoveImage(index); }}>&times;</button>
                                      </div>
                                  ))}
                              </div>
                          )}
                          {errors.images && <p className={styles.errorText} style={{textAlign: 'center'}}>{errors.images}</p>}
                      </div>

                      {/* CỘT PHẢI FORM */}
                      <div className={styles.formFieldsSection}>
                          <h3 className={styles.sectionTitle}>Thông tin sản phẩm</h3>
                          
                          {selectedCategory && (
                                  <SpecificFields
                                      fields={selectedCategory.specificFields}
                                      formData={formData}
                                      handleChange={handleChange}
                                      errors={errors}
                                      isDescriptionGuideVisible={isDescriptionGuideVisible}
                                      onFocus={(fieldName) => {
                                          if (fieldName === 'description') setIsDescriptionGuideVisible(true);
                                      }}
                                      onBlur={(fieldName) => {
                                          if (fieldName === 'description') setIsDescriptionGuideVisible(false);
                                      }}
                                  />
                              )}
                              
                          <hr className={styles.sectionDivider} />
                          <h3 className={styles.sectionTitle}>Địa chỉ</h3>

                          <div className={styles.inputGroup}>
                              <label className={styles.label}>Tỉnh/Thành phố *</label>
                              <select 
                                className={`${styles.input} ${styles.select}`} 
                                onChange={handleProvinceChange} 
                                required 
                                value={selectedProvinceCode || ""}
                                data-testid="post-province-select"
                              >
                                  <option value="">-- Chọn Tỉnh/Thành phố --</option>
                                  {provinces.map((p) => (<option key={p.code} value={p.code}>{p.name}</option>))}
                              </select>
                              {errors.province && <p className={styles.errorText}>{errors.province}</p>}
                          </div>
                          <div className={styles.inputGroup}>
                              <label className={styles.label}>Quận/Huyện *</label>
                              <select 
                                className={`${styles.input} ${styles.select}`} 
                                onChange={handleDistrictChange} 
                                required 
                                disabled={!selectedProvinceCode} 
                                value={selectedDistrictCode || ""}
                                data-testid="post-district-select"
                              >
                                  <option value="">-- Chọn Quận/Huyện --</option>
                                  {districts.map((d) => (<option key={d.code} value={d.code}>{d.name}</option>))}
                              </select>
                              {errors.district && <p className={styles.errorText}>{errors.district}</p>}
                          </div>
                          <div className={styles.inputGroup}>
                              <label className={styles.label}>Phường/Xã *</label>
                              <select 
                                className={`${styles.input} ${styles.select}`} 
                                onChange={handleWardChange} 
                                required 
                                disabled={!selectedDistrictCode} 
                                value={formData.ward || ""}
                                data-testid="post-ward-select"
                              >
                                  <option value="">-- Chọn Phường/Xã --</option>
                                  {wards.map(w => <option key={w.code} value={w.name}>{w.name}</option>)}
                              </select>
                              {errors.ward && <p className={styles.errorText}>{errors.ward}</p>}
                          </div>
                          <div className={styles.inputGroup}>
                              <label className={styles.label}>Địa chỉ cụ thể (Không bắt buộc)</label>
                              <input 
                                className={styles.input} 
                                type="text" 
                                name="streetAddress" 
                                value={formData.streetAddress || ''} 
                                onChange={handleChange} 
                                placeholder="VD: Số nhà 123, đường ABC, ngõ 45" 
                                data-testid="post-street-address-input"
                              />
                          </div>
                      </div>
                  </div>

                  {/* --- THANH ĐIỀU KHIỂN DƯỚI CÙNG --- */}
                  <div className={styles.footerActions}>
                      <div 
                        className={styles.postTypeSelector} 
                        onClick={() => setIsPostTypeModalOpen(true)}
                        data-testid="post-type-selector"
                      >
                          <span>{selectedPostType.label}</span>
                          <MdKeyboardArrowDown />
                      </div>
                      <div className={styles.submitActions}>
                          <button 
                            type="button" 
                            onClick={handlePreview} 
                            className={styles.secondaryButton}
                            data-testid="post-preview-button"
                          >Xem trước</button>
                          
                          <button 
                            type="button" 
                            onClick={handleSaveDraft} 
                            disabled={draftLoading} 
                            className={styles.secondaryButton}
                            data-testid="post-save-draft-button"
                          >{draftLoading ? 'Đang lưu...' : 'Lưu nháp'}</button>
                          
                          <button 
                            type="submit" 
                            disabled={loading} 
                            className={styles.submitButton}
                            data-testid="post-submit-button"
                          >{loading ? "Đang Đăng..." : "Đăng Tin"}</button>
                      </div>
                  </div>
              </form>
          </div>
      </>
  );}

    export default CreatePost;
